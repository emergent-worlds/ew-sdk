cmake_minimum_required(VERSION 3.16)
cmake_policy(SET CMP0091 NEW)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded")
project(ew_native_sdk_sample LANGUAGES C)
if(CMAKE_CROSSCOMPILING OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Use the actual matching 64-bit worker.")
endif()
if(NOT CMAKE_SYSTEM_PROCESSOR MATCHES "^(x86_64|AMD64|amd64)$")
  message(FATAL_ERROR "This finite portable sample target requires an actual x64 worker.")
endif()
if(NOT WIN32 AND NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
  message(FATAL_ERROR "macOS continues to use its unchanged Swift/JSC program.")
endif()
if(NOT DEFINED EW_EMBEDDED_DIR OR NOT EXISTS "${EW_EMBEDDED_DIR}/embedded.h")
  message(FATAL_ERROR "The exact full source bundle is required.")
endif()
# Compile only the five upstream engine-core sources. No CLI, std/os or native module loader.
set(QJS "${CMAKE_CURRENT_LIST_DIR}/../../vendor/quickjs-ng")
add_library(ew_quickjs STATIC "${QJS}/cutils.c" "${QJS}/dtoa.c" "${QJS}/libregexp.c" "${QJS}/libunicode.c" "${QJS}/quickjs.c")
set_target_properties(ew_quickjs PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_include_directories(ew_quickjs PUBLIC "${QJS}")
target_compile_definitions(ew_quickjs PRIVATE _GNU_SOURCE)
if(WIN32)
  target_compile_definitions(ew_quickjs PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
  if(MSVC)
    target_compile_options(ew_quickjs PRIVATE /experimental:c11atomics /utf-8)
  endif()
else()
  find_package(Threads REQUIRED)
  target_link_libraries(ew_quickjs PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
add_executable(ew_native main.c channel.c)
set_target_properties(ew_native PROPERTIES OUTPUT_NAME "KestrelYards" C_STANDARD 11 C_STANDARD_REQUIRED YES)
target_include_directories(ew_native PRIVATE "${EW_EMBEDDED_DIR}")
target_link_libraries(ew_native PRIVATE ew_quickjs)
if(WIN32)
  target_compile_definitions(ew_native PRIVATE WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0601)
  target_link_libraries(ew_native PRIVATE ws2_32 bcrypt)
  if(MSVC)
    target_compile_options(ew_native PRIVATE /W4 /utf-8 /experimental:c11atomics)
    target_link_options(ew_native PRIVATE /STACK:8388608 /DYNAMICBASE /NXCOMPAT)
  endif()
else()
  target_compile_definitions(ew_native PRIVATE _GNU_SOURCE)
  target_compile_options(ew_native PRIVATE -Wall -Wextra -Wformat=2)
  target_link_libraries(ew_native PRIVATE m)
endif()
